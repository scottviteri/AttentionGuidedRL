\begin{abstract}
This document provides a mathematical formulation of the Attention-Guided Reinforcement Learning system, where a language model (Llama-3.2-3B or GPT-2) learns to autonomously sequence its own training data. The key innovation is a \textbf{chain rule policy gradient} approach that combines:

\begin{enumerate}
    \item \textbf{Direct reward optimization}: $\nabla_\theta r_t$ - optimizing the reward function itself
    \item \textbf{Trajectory-average policy gradients}: $\bar{R}(\tau) \nabla_\theta \log \pi_\theta(a_t | s_t)$ - uniform credit assignment
\end{enumerate}

The model generates vector queries using attention mechanisms and selects key-value pairs from Wikipedia articles to maximize learning progress. Unlike standard REINFORCE \cite{williams1992simple}, this approach optimizes both the policy and the reward function simultaneously through differentiable reward computation, creating a more sophisticated learning dynamic.
\end{abstract}

\section{Problem Formulation}

\subsection{State Space and Action Space}

\begin{definition}[State Space]
At timestep $t$, the state $s_t$ consists of:
\begin{align}
s_t = (c_t, \mathcal{K}_t^{\text{available}})
\end{align}
where:
\begin{itemize}
    \item $c_t \in \mathbb{Z}^{L_t}$ is the context sequence of length $L_t$
    \item $\mathcal{K}_t^{\text{available}} \subseteq \{1, 2, \ldots, K\}$ is the set of available key indices
\end{itemize}
\end{definition}

\begin{definition}[Action Space]
The action space $\mathcal{A}_t$ at timestep $t$ is the set of available key indices:
\begin{align}
\mathcal{A}_t = \mathcal{K}_t^{\text{available}}
\end{align}
\end{definition}

\subsection{Trajectory and Episode Structure}

A trajectory $\tau$ consists of $T$ steps:
\begin{align}
\tau = \{(s_1, a_1, r_1), (s_2, a_2, r_2), \ldots, (s_T, a_T, r_T)\}
\end{align}

where $T$ is the number of key-value pairs per episode (NUM\_KV\_PAIRS in the code).

\section{Policy Definition}

\subsection{Vector Query Generation}

The policy $\pi_\theta$ operates through a vector query mechanism parameterized by LoRA adapter weights $\theta$ \cite{hu2021lora}.

\begin{definition}[Query Vector Generation]
Given context $c_t$, the query vector is generated as:
\begin{align}
q_t = \text{Embed}_\theta(c_t \oplus [\text{``Query''}])_{-1}
\end{align}
where:
\begin{itemize}
    \item $\text{Embed}_\theta(\cdot)$ extracts embeddings from the second-to-last attention layer
    \item $\oplus$ denotes sequence concatenation
    \item $(\cdot)_{-1}$ extracts the embedding of the last token (the ``Query'' token)
    \item $q_t \in \mathbb{R}^{d_{\text{model}}}$ where $d_{\text{model}}$ is the model dimension
\end{itemize}
\end{definition}

This focused extraction ensures that the model learns to encode its query intent specifically into the ``Query'' token, rather than diluting the signal across the entire context.

\subsection{Multi-Head Attention Similarity}

The system handles both standard Multi-Head Attention (MHA) and Grouped Query Attention (GQA) \cite{shazeer2019fast}.

\begin{definition}[Head-wise Similarity Computation]
For each attention head $h \in \{1, 2, \ldots, H\}$ and key $k$:
\begin{align}
\text{sim}_{t,k}^{(h)} = \frac{(q_t^{(h)})^T k_k^{(\text{group}(h))}}{\sqrt{d_{\text{head}}}} \cdot \frac{1}{\tau}
\end{align}
where:
\begin{itemize}
    \item $q_t^{(h)} \in \mathbb{R}^{d_{\text{head}}}$ is the query for head $h$
    \item $k_k^{(\text{group}(h))} \in \mathbb{R}^{d_{\text{head}}}$ is the key for the corresponding group
    \item $\text{group}(h) = \lfloor h \cdot G / H \rfloor$ maps heads to groups (for GQA)
    \item $G$ is the number of key-value groups
    \item $H$ is the number of query heads
    \item $d_{\text{head}} = d_{\text{model}} / H$ is the head dimension
    \item $\tau = 1.0$ is the temperature parameter
\end{itemize}
\end{definition}

\begin{definition}[Per-Head Probability Distribution]
For each head $h$, the probability distribution over keys is:
\begin{align}
p_{t,k}^{(h)} = \frac{\exp(\text{sim}_{t,k}^{(h)})}{\sum_{k' \in \mathcal{K}_t^{\text{available}}} \exp(\text{sim}_{t,k'}^{(h)})}
\end{align}
\end{definition}

\begin{definition}[Policy Distribution]
The final policy distribution is obtained by averaging over heads:
\begin{align}
\pi_\theta(k | s_t) = \frac{1}{H} \sum_{h=1}^{H} p_{t,k}^{(h)}
\end{align}
This preserves the probability distribution property: $\sum_{k \in \mathcal{K}_t^{\text{available}}} \pi_\theta(k | s_t) = 1$.
\end{definition}

\section{Reward Function}

\begin{definition}[Step Reward]
The reward at step $t$ is the per-token average log probability:
\begin{align}
r_t = \frac{1}{|v_t|} \sum_{i=1}^{|v_t|} \log p_\theta\big(v_{t,i} \mid c_t, k_t, v_{t,<i}\big)
\end{align}
where:
\begin{itemize}
    \item $v_t$ are the value tokens at step $t$
    \item $k_t$ are the selected key tokens at step $t$
    \item $p_\theta(\cdot)$ is the current adapter model probability
    \item $v_{t,i}$ is the $i$-th token and $v_{t,<i}$ represents preceding tokens
\end{itemize}
\end{definition}

\textbf{Configuration Option:} The system supports two reward modes:
\begin{itemize}
    \item \texttt{subtract\_base\_model\_logprobs=False} (default): Uses the formula above
    \item \texttt{subtract\_base\_model\_logprobs=True}: Subtracts reference model log probabilities:
\end{itemize}

\begin{align}
r_t = \frac{1}{|v_t|} \sum_{i=1}^{|v_t|} \Big( \log p_\theta(v_{t,i} \mid \cdot) - \log p_{\text{ref}}(v_{t,i} \mid \cdot) \Big)
\end{align}



\section{Optimization Objective}

\begin{theorem}[Trajectory-Average Policy Gradient]
The system maximizes:
\begin{align}
\mathcal{J}(\theta) = \mathbb{E}_{\tau \sim \pi_\theta} \left[ \bar{R}(\tau) \right]
\end{align}

where $\bar{R}(\tau) = \frac{1}{T} \sum_{t=1}^T r_t$ is the trajectory average reward.

\textbf{Chain Rule Gradient Derivation:} Since both policy and rewards depend on $\theta$, the gradient is:
\begin{align}
\nabla_\theta \mathcal{J}(\theta) = \mathbb{E}_{\tau \sim \pi_\theta} \left[ \frac{1}{T} \sum_{t=1}^T \nabla_\theta r_t + \bar{R}(\tau) \sum_{t=1}^T \nabla_\theta \log \pi_\theta(a_t | s_t) \right]
\end{align}

This gives two terms:
\begin{enumerate}
    \item \textbf{Reward gradient term}: $\frac{1}{T} \sum_{t=1}^T \nabla_\theta r_t$ - direct optimization of reward function
    \item \textbf{Policy gradient term}: $\bar{R}(\tau) \sum_{t=1}^T \nabla_\theta \log \pi_\theta(a_t | s_t)$ - trajectory-average weighted policy gradient
\end{enumerate}

Each action receives uniform credit: the trajectory average reward $\bar{R}(\tau)$.
\end{theorem}

\section{Training Algorithm}

\textbf{Loss Function:} The training implements the full chain rule gradient:

\begin{align}
\mathcal{L}(\theta) = \mathbb{E}_{\text{batch}} \left[ -\frac{1}{T} \sum_{t=1}^T r_t - \bar{R}(\tau) \sum_{t=1}^T \log \pi_\theta(a_t | s_t) \right]
\end{align}

where:
\begin{itemize}
    \item $\bar{R}(\tau) = \frac{1}{T} \sum_{t=1}^T r_t$ is the trajectory average reward
    \item The first term optimizes rewards directly via $\nabla_\theta r_t$
    \item The second term is the trajectory-average weighted policy gradient
\end{itemize}

\begin{algorithm}
\caption{Chain Rule Policy Gradient Training}
\begin{algorithmic}[1]
\REQUIRE Base model, tokenizer, Wikipedia dataset
\STATE Initialize LoRA adapter parameters $\theta$
\FOR{each training episode}
    \STATE Sample Wikipedia article and extract key-value pairs
    \STATE Generate trajectory $\tau$ using current policy $\pi_\theta$
    \STATE Compute trajectory average reward $\bar{R}(\tau)$
    \STATE Compute loss and update parameters via gradient descent
\ENDFOR
\end{algorithmic}
\end{algorithm}

\textbf{Key Properties:}
\begin{itemize}
    \item Each action receives uniform credit: the trajectory average reward
    \item No baseline subtraction or value function needed
    \item Simple implementation with uniform credit assignment
\end{itemize}

\section{Implementation Details}

\subsection{Model Architecture}
\begin{itemize}
    \item \textbf{Base Models}: Llama-3.2-3B or GPT-2
    \item \textbf{LoRA Configuration}: rank=8, $\alpha$=16, dropout=0.0
    \item \textbf{Precision}: bfloat16 for computation, float32 for similarity calculations
    \item \textbf{Context Window}: Up to model maximum (2048 for GPT-2, 4096+ for Llama)
\end{itemize}

\subsection{Training Configuration}
\begin{itemize}
    \item \textbf{Optimizer}: AdamW with learning rate $5 \times 10^{-4}$
    \item \textbf{Batch Size}: 4 trajectories per update
    \item \textbf{Gradient Clipping}: Norm clipped to 1.0
    \item \textbf{Episodes}: 10,000 total training episodes
    \item \textbf{Checkpoint Frequency}: Every 100 episodes
\end{itemize}

\subsection{Data Processing}
\begin{itemize}
    \item \textbf{Dataset}: Wikipedia 2022-03-01 English subset
    \item \textbf{Token Counts}: 10 tokens per key, 10 tokens per value
    \item \textbf{Trajectory Length}: Up to 10 key-value pairs per episode
    \item \textbf{Prefixes}: "Key: " and "Value: " for context building
\end{itemize}

\section{Mathematical Properties}

The trajectory-average approach gives uniform credit assignment where each action receives the same weight: the trajectory average reward $\bar{R}(\tau)$. This treats all actions in a trajectory equally, following standard REINFORCE convergence guarantees \cite{williams1992simple}.

\section{Conclusion}

This mathematical formulation describes a reinforcement learning system where a language model learns to sequence its own training data using:

\begin{enumerate}
    \item \textbf{Chain rule policy gradients}: Simultaneous optimization of both policy and reward function
    \item \textbf{Vector-based policy}: Query generation from attention embeddings
    \item \textbf{Multi-head attention similarity}: For key selection across MHA/GQA architectures
    \item \textbf{Trajectory-average credit assignment}: Uniform weighting across all actions
    \item \textbf{Differentiable reward computation}: Enabling direct reward optimization
\end{enumerate}

The key innovation is the \textbf{chain rule approach} that goes beyond standard REINFORCE \cite{williams1992simple} by optimizing the reward function directly ($\nabla_\theta r_t$) while using trajectory-average rewards for policy gradients ($\bar{R}(\tau) \nabla_\theta \log \pi_\theta$). This creates a sophisticated learning dynamic where the model simultaneously improves its policy and the rewards it receives.
